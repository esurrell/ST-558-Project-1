get_var_values <- function(year){

  vars <- GET(paste0("https://api.census.gov/data/", 
                   year, 
                   "/acs/acs1/pums/variables.json"))

  vars <- fromJSON(rawToChar(vars$content))

  if (year == 2021 | year == 2022){
    vars <- list(
      AGEP = vars$variables$AGEP$values,
      GASP = vars$variables$GASP$values,
      GRPIP = vars$variables$GRPIP$values,
      JWAP = vars$variables$JWAP$values,
      JWDP = vars$variables$JWDP$values,
      JWMNP = vars$variables$JWMNP$values,
      PWGTP = vars$variables$PWGTP$values,
      FER = vars$variables$FER$values,
      HHL = vars$variables$HHL$values,
      SCH = vars$variables$SCH$values,
      SCHL = vars$variables$SCHL$values,
      SEX = vars$variables$SEX$values,
      ST = vars$variables$ST$values,
      REGION = vars$variables$REGION$values,
      DIVISION = vars$variables$DIVISION$values
      )
  }
  
  else if (year == 2023 | year == 2024){
    vars <- list(
      AGEP = vars$variables$AGEP$values,
      GASP = vars$variables$GASP$values,
      GRPIP = vars$variables$GRPIP$values,
      JWAP = vars$variables$JWAP$values,
      JWDP = vars$variables$JWDP$values,
      JWMNP = vars$variables$JWMNP$values,
      PWGTP = vars$variables$PWGTP$values,
      FER = vars$variables$FER$values,
      HHL = vars$variables$HHL$values,
      SCH = vars$variables$SCH$values,
      SCHL = vars$variables$SCHL$values,
      SEX = vars$variables$SEX$values,
      STATE = vars$variables$STATE$values, #STATE is used instead of ST
      REGION = vars$variables$REGION$values,
      DIVISION = vars$variables$DIVISION$values
      )
    }
  
  return(vars)
}

Start with a helper function to take what is returned by GET() and turn it into a nice tibble:

parse_from_get <- function(getobject){
  
  #parse data:
  content <- fromJSON(rawToChar(getobject$content))
  
  #remove header row:
  content <- content[-1,]
  
  #make numeric:
  content <- matrix(as.numeric(content), ncol = ncol(content))

  #format as tibble and return:
  return(as_tibble(content))
}

Write a second helper function to convert variables into factors:

convert_to_factor <- function(tibble, year, variable){
  rds <- read_rds("data/variables.rds")
  values <- rds[[paste0("vars", year)]]
  values <- values[[variable]]
  values <- values$item
  
  cleaned <- data.frame(rownum = 1:length(as.numeric(names(values))))

  for (i in 1:length(as.numeric(names(values)))){
    cleaned$num[i] <- as.numeric(names(values))[i]
    cleaned$value[i] <- values[[i]]
  }

  tibble[[variable]] <- factor(tibble[[variable]],
                      cleaned$num, 
                      cleaned$value)
  return(tibble)
}

Use these helper functions inside of larger function which queries the API:

query <- function(year, nvar, cvar, geolevel, geovalue){
  
  #translate geolevel into numeric geolevel:
  if (geolevel == "STATE"){geolevel_translated <- "040"}
  else if (geolevel == "REGION"){geolevel_translated <- "020"}
  else if (geolevel == "DIVISION"){geolevel_translated < "030"}
  
  #API call:
  raw_api <- GET(paste0("https://api.census.gov/data/",
    year,
    "/acs/acs1/pums?get=",
    paste(nvar, collapse = ","),
    ",",
    paste(cvar, collapse = ","),
    "&ucgid=",
    geolevel_translated,
    "0000US",
    geovalue,
    "&key=",
    key))

  #Parse:
  tibble <- parse_from_get(raw_api)

  #Fix column names:
  colnames(tibble) <- c(nvar, cvar, geolevel)

  #Convert character variables and geolevel to factors:
  for (i in c(cvar, geolevel)){
    tibble <- tibble |>
      convert_to_factor(year, i)
  }
  
  class(tibble) <- c("census", class(tibble))
  
  return(tibble)
}

Combining Years

Write a function that calls the single-year function as many times as needed and then combines the data into one final tibble with a year variable.

multiyear_query <- function(years, nvar, cvar, geolevel, geovalue){
  
  firstyear <- 1
  
  for (year in years){
    
    if(firstyear==1){
      tibble <- query(year, nvar, cvar, geolevel, geovalue) |>
        mutate(year = year)
  
      firstyear <- 0
    }
    else {
      tibble <- dplyr::bind_rows(tibble,
      query(year, nvar, cvar, geolevel, geovalue) |>
        mutate(year = year))
    }
  }
  
  return(tibble)
}
